% ---------- step 1 ----------
\renewcommand{\storychapter}{1 \,$\cdot$\, The price level}
\qslide{A1: Is the average espresso different from 3 USD?}{A1_figure.pdf}
  {64.7 CZK}{average espresso \,$\cdot$\, 95\% CI [63.0, 66.5] \,$\cdot$\, $n=199$}
  {Benchmark: 65.3 CZK (21.76 CZK per USD)\\ Effect size: $d=-0.05$ (negligible)\\ One-sample $t$-test, $p=.503$}
  {No evidence that the average differs from 3 USD.}

\qslide{A2: Do roasteries charge more for an espresso?}{A2_figure.pdf}
  {+3.3 CZK}{roasteries minus others \,$\cdot$\, 95\% CI [$-1.1$, 7.6]}
  {Effect size: $d=0.26$ (small)\\ Welch $t$-test, one-tailed $p=.070$\\ 23 roasteries, 176 other caf\'es}
  {No clear evidence of a premium. With only 23 roasteries the test is weak.}

\qslide{A3: How much more is a flat white than a cappuccino?}{A3_figure.pdf}
  {+12.2 CZK}{in the same caf\'e \,$\cdot$\, 95\% CI [10.9, 13.6] \,$\cdot$\, $n=197$}
  {Effect size: $d_z=1.29$ (very large)\\ Paired $t$-test, one-tailed $p<.001$}
  {About 15\% more than a cappuccino. Our clearest result.}

% ---------- step 2 ----------
\renewcommand{\storychapter}{2 \,$\cdot$\, Who charges more?}
\qslide{A4: Does the cappuccino price differ by caf\'e type?}{A4_figure.pdf}
  {80.4 / 86.0 / 97.0 CZK}{independent / local chain / international chain}
  {Welch ANOVA: $F(2,\,22.19)=4.62$, $p=.021$\\ Effect: independent vs.\ local chain $-5.6$ CZK, 95\% CI [$-10.8$, $-0.4$], $p=.030$}
  {Local chains charge about 7\% more. Only 10 international chains: too few to say more.}

\qslide{A5: Prague 1 and an above-median flat white}{A5_figure.pdf}
  {+27.7 percentage points}{50.7\% in Prague 1 vs.\ 23.1\% elsewhere \,$\cdot$\, 95\% CI [13.7, 41.7]}
  {Effect size: Cram\'er's $V=.28$ (moderate)\\ $\chi^2$ test, one-tailed $p<.001$\\ Median flat white: 95 CZK}
  {Prague 1 caf\'es are much more likely to price a flat white high.}

% ---------- step 3 ----------
\renewcommand{\presenter}{Eden}
\renewcommand{\storychapter}{3 \,$\cdot$\, Do ratings explain price?}
\qslide{A6: Is cappuccino price correlated with rating?}{A6_figure.pdf}
  {$r=-.22$}{Pearson correlation \,$\cdot$\, 95\% CI [$-.35$, $-.08$] \,$\cdot$\, $n=195$}
  {Effect size: $r^2=.05$ (5\% shared variance)\\ $p=.002$}
  {Weak and negative: higher-rated caf\'es charge slightly less, not more.}

\qslide{A7: How much does price change per rating point?}{A7_figure.pdf}
  {$-16.3$ CZK per point}{95\% CI [$-26.4$, $-6.2$] \,$\cdot$\, about $-1.6$ CZK per 0.1}
  {Effect size: $R^2=.05$\\ Linear regression, $p=.002$}
  {A clear but small association: ratings explain only 5\% of the variation in price.}

\qslide{A8: Does it hold once espresso price is accounted for?}{A8_figure.pdf}
  {$-2.7$ CZK per point}{95\% CI [$-12.4$, 7.0] \,$\cdot$\, only 16.6\% of the slope remains}
  {Effect size: $R^2$ rises from .05 to .68\\ Multiple regression, $p=.584$\\ Espresso price coefficient: 0.97}
  {The rating link mostly reflects espresso prices: higher-rated caf\'es tend to have cheaper espresso.}

\qslide{A9: Does the rating--price slope differ for roasteries?}{A9_figure.pdf}
  {$-11.3$ CZK}{slope difference, roasteries minus others \,$\cdot$\, 95\% CI [$-45.9$, 23.3]}
  {Slopes: other caf\'es $-16.0$, roasteries $-27.3$ CZK per point\\ Effect size: $R^2=.06$\\ Regression with interaction, $p=.519$}
  {No evidence the relationship differs between roasteries and other caf\'es.}

\qslide{A10: Are better-rated caf\'es more likely to charge above the median?}{A10_figure.pdf}
  {$-6.2$ pp per 0.1 rating}{average marginal effect \,$\cdot$\, 95\% CI [$-9.2$, $-3.2$]}
  {Effect size: odds ratio $=0.74$\\ Logistic regression, one-tailed $p>.999$}
  {No: the effect goes the opposite way. This direction was not predicted, so it is exploratory.}

% ---------- conclusion ----------

\renewcommand{\storychapter}{Conclusion}
\begin{frame}{What we found}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \infocard{Prices}{An average espresso costs about what 3 USD buys. A flat white costs about 12 CZK more than a cappuccino.}
      \par\vspace{6pt}
      \infocard{Where and who}{Prague 1 caf\'es are far more likely to charge above the median. Local chains charge a little more than independent caf\'es.}
    \column{0.48\textwidth}
      \infocard{Ratings}{Higher ratings do not mean higher prices. The weak negative link mostly follows espresso prices.}
      \par\vspace{6pt}
      \infocard{Roasteries}{No clear evidence that roasteries charge more.}
  \end{columns}
  \vspace{8pt}
  \begin{center}\takecard{Limits: convenience sample, observational data, caf\'es in the same district are not independent, small roastery and chain groups.}\end{center}
\end{frame}
