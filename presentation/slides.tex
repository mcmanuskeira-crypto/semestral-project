% ---------- A1 ----------
\begin{frame}{A1 \quad Is espresso more or less expensive than 3 USD?}
  \result{64.7 CZK average espresso price}
  \vspace{.45cm}
  \begin{itemize}
    \item Benchmark: \textbf{65.3 CZK} (equivalent of 3 USD).
    \item Difference: only \textbf{-0.6 CZK}.
    \item 95\% CI: [63.0, 66.5]; \quad $p=.503$; \quad $n=199$.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} No evidence that the sample's mean espresso price differs from the 3 USD benchmark.
  \end{beamercolorbox}
\end{frame}

% ---------- A2 ----------
\begin{frame}{A2 \quad Do roasteries charge more for espresso?}
  \result{+3.3 CZK at roasteries, on average}
  \vspace{.45cm}
  \begin{itemize}
    \item Difference (roasteries minus other cafés): \textbf{3.3 CZK}.
    \item 95\% CI: [-1.1, 7.6]; \quad one-tailed $p=.070$.
    \item 23 roasteries and 176 other cafés.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} The estimated premium is small and not statistically significant. The result does not rule out a modest premium.
  \end{beamercolorbox}
\end{frame}

% ---------- A3 ----------
\begin{frame}{A3 \quad How much more does a flat white cost?}
  \result{12.2 CZK more than a cappuccino}
  \vspace{.45cm}
  \begin{itemize}
    \item Average difference within the same café.
    \item 95\% CI: [10.9, 13.6]; \quad one-tailed $p<.001$.
    \item 197 cafés had both prices.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} Flat whites were consistently more expensive—about 15\% more than the average cappuccino price.
  \end{beamercolorbox}
\end{frame}

% ---------- A4 ----------
\begin{frame}{A4 \quad Does cappuccino price differ by café type?}
  \result{Local chains charged 5.6 CZK more than independents}
  \vspace{.35cm}
  \begin{center}
  \begin{tabular}{lrr}
    \toprule
    Café type & Mean price & Cafés ($n$)\\
    \midrule
    Independent & 80.4 CZK & 138\\
    Local chain & 86.0 CZK & 50\\
    International chain & 97.0 CZK & 10\\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{.2cm}
  \begin{itemize}
    \item Overall difference: Welch ANOVA, $p=.021$.
    \item Only independent vs. local chain was significant (adjusted $p=.030$).
  \end{itemize}
  \begin{beamercolorbox}[rounded=true,sep=.6em]{block body}
    \textbf{Takeaway:} Local chains were modestly more expensive than independent cafés. International-chain estimates are uncertain (only 10 cafés).
  \end{beamercolorbox}
\end{frame}

% ---------- A5 ----------
\begin{frame}{A5 \quad Are above-median flat whites more common in Prague 1?}
  \result{50.7\% in Prague 1 \quad vs. \quad 23.1\% elsewhere}
  \vspace{.45cm}
  \begin{itemize}
    \item Difference: \textbf{27.7 percentage points}.
    \item 95\% CI: [13.7, 41.7]; \quad one-tailed $p<.001$.
    \item “Above median” means a flat white price strictly above 95 CZK.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} Cafés in Prague 1 were substantially more likely to charge above the sample median for a flat white.
  \end{beamercolorbox}
\end{frame}

% ---------- A6 ----------
\begin{frame}{A6 \quad Is cappuccino price related to Adjusted Rating?}
  \result{A weak negative correlation: $r=-.22$}
  \vspace{.45cm}
  \begin{itemize}
    \item 95\% CI: [-.35, -.08].
    \item $p=.002$; \quad $n=195$.
    \item Only about 5\% of price variation was shared with Adjusted Rating.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} Higher-rated cafés tended to charge slightly less for cappuccino, but the relationship was weak.
  \end{beamercolorbox}
\end{frame}

% ---------- A7 ----------
\begin{frame}{A7 \quad How does cappuccino price change with Adjusted Rating?}
  \result{-16.3 CZK per 1-point increase in rating}
  \vspace{.45cm}
  \begin{itemize}
    \item 95\% CI: [-26.4, -6.2].
    \item $p=.002$; \quad $R^2=.05$; \quad $n=195$.
    \item A more realistic 0.1-point increase corresponds to about \textbf{1.6 CZK lower}.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} The negative association is statistically clear but small in practical terms; the model explains only 5\% of price variation.
  \end{beamercolorbox}
\end{frame}

% ---------- A8 ----------
\begin{frame}{A8 \quad Does the rating link remain after accounting for espresso price?}
  \result{Only 16.6\% of the original slope remained}
  \vspace{.4cm}
  \begin{itemize}
    \item Rating slope changed from \textbf{-16.3} to \textbf{-2.7 CZK} per point.
    \item 95\% CI: [-12.4, 7.0]; \quad $p=.584$.
    \item Model $R^2$ increased from .05 to .68 after adding espresso price.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} The rating–cappuccino relationship was no longer statistically clear once espresso price was included.
  \end{beamercolorbox}
\end{frame}

% ---------- A9 ----------
\begin{frame}{A9 \quad Does the rating–price link differ for roasteries?}
  \result{No clear difference between the two groups}
  \vspace{.4cm}
  \begin{itemize}
    \item Estimated slope: other cafés \textbf{-16.0} CZK per point.
    \item Estimated slope: roasteries \textbf{-27.3} CZK per point.
    \item Difference in slopes: -11.3 CZK; 95\% CI [-45.9, 23.3]; \quad $p=.519$.
    \item Only 23 roasteries were included.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} Although the estimated slope was steeper for roasteries, the difference was too uncertain to distinguish from zero.
  \end{beamercolorbox}
\end{frame}

% ---------- A10 ----------
\begin{frame}{A10 \quad Are better-rated cafés more likely to price flat whites above the median?}
  \result{Estimated probability fell by 6.2 percentage points per 0.1 rating point}
  \vspace{.35cm}
  \begin{itemize}
    \item 95\% CI: [-9.17, -3.21]; \quad $n=194$.
    \item Odds ratio: 0.74 (95\% CI [0.62, 0.87]).
    \item The estimated direction was opposite to the prediction; one-tailed $p> .999$.
  \end{itemize}
  \vfill
  \begin{beamercolorbox}[rounded=true,sep=.7em]{block body}
    \textbf{Takeaway:} No support for the predicted positive relationship. The negative direction is exploratory, not a confirmed finding.
  \end{beamercolorbox}
\end{frame}

% ---------- Final ----------
\begin{frame}{What do the results tell us overall?}
  \begin{columns}[T,onlytextwidth]
    \column{.49\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=.8em,wd=\linewidth]{block body}
        \textbf{\color{DeepRose}Clearer patterns}
        \begin{itemize}
          \item Flat whites cost about 12 CZK more than cappuccinos.
          \item Prague 1 cafés were more likely to charge above-median flat white prices.
          \item Local chains charged more than independent cafés for cappuccino.
        \end{itemize}
      \end{beamercolorbox}
    \column{.49\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=.8em,wd=\linewidth]{block body}
        \textbf{\color{Mocha}More uncertain patterns}
        \begin{itemize}
          \item No clear espresso premium for roasteries.
          \item Rating–price association weakened after controlling for espresso price.
          \item No clear difference in rating slopes between roasteries and other cafés.
        \end{itemize}
      \end{beamercolorbox}
  \end{columns}
  \vfill
  {\small\bfseries Important limitations}
  \small The data are observational and come from a convenience sample. Findings describe the cafés sampled; they cannot be generalised to all Prague cafés or interpreted as causal effects.
  \vspace{.25cm}
